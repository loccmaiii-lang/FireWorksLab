   ```js
   window.FWLImporter = {
     version: '5.0.0',
     status(),                    // → {ue: 'connected' | 'disconnected' | 'unavailable', detail, retryInSec}
     reconnect(),                 // UE 断开时「重新连接」调这个
     scan(dirHandle),             // 读导出文件夹（烘焙器交付写的 df.firework-resource/1 索引）→ 包列表
                                  //   [{id, name, revision, platform, exportedAt, thumb, state: '待导入' | '有新版本' | '已导入',
                                  //     assets: [{name, kind, uePath, action: '新建' | '覆盖' | '不变'}]}]
     precheck(pkgId),             // → {ok, newCount, overwriteCount, issues: [{level, text}]}
     importPkg(pkgId, {onProgress}),   // 用户点「导入」才写 UE → {ok, written, failed: [{name, reason}]}
     on(event, cb),               // 'status' | 'progress'
   };
   ```
